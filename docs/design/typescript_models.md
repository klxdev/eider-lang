# Models with TypeScript Export

Write the models once, then compile:

```
eider build --target wasm32 --emit ts
```

This produces three files:

- `warehouse.wasm` contains the compiled logic.
- `warehouse.js` loads the module and converts values at the boundary.
- `warehouse.d.ts` holds the generated TypeScript types.

## 1. The models (source of truth)

```eider
// src/models.ei
module warehouse.models

import std.time.Instant

export(ts) type Sku = distinct str

export(ts) type Money = { cents: i64 }

export(ts) type Item = {
  sku: Sku,
  name: str,
  price: Money,
  grams: u32,
}

export(ts) type Line = { sku: Sku, qty: u32 }

export(ts) type BoxSize = | Small | Medium | Large

export(ts) type OrderStatus =
  | Pending
  | Picking(picker: u32, started: Instant)
  | Packed(box: BoxSize)
  | Shipped(tracking: str)
  | Cancelled(reason: str)

export(ts) type Order = {
  id: u64,
  lines: List[Line],
  status: OrderStatus = Pending,
  priority: u8 = 0,
}

export(ts) type Event =
  | StartPick(picker: u32)
  | Pack
  | Ship(tracking: str)
  | Cancel(reason: str)

export(ts) type FulfillError =
  | UnknownSku(Sku)
  | OutOfStock(sku: Sku, wanted: u32, have: u32)
  | TooHeavy(grams: u32)
  | InvalidTransition(from: OrderStatus, event: str)

export(ts) type Inventory = {
  items: Map[Sku, Item],
  stock: Map[Sku, u32],
}
```

## 2. Exported logic

`inout` can't be expressed in TypeScript, so exported functions take values and return new values. Your own code keeps using `inout` internally.

```eider
// src/api.ei
module warehouse.api

import warehouse.models.*
import warehouse.fulfillment as f

export(ts) fn advance(order: Order, event: Event, inv: Inventory)
  -> Result[Order, FulfillError]
  uses clock
{
  let mut o = order.copy()
  f.advance(inout o, event, inv)?
  return Ok(o)
}

export(ts) fn reserve_all(inv: Inventory, lines: List[Line])
  -> Result[Inventory, FulfillError]
{
  let mut next = inv.copy()
  next.reserve_all(lines)?
  return Ok(next)
}

export(ts) fn choose_box(grams: u32) -> Result[BoxSize, FulfillError] {
  return f.choose_box(grams)
}

// Generics must be instantiated before export
export(ts) fn total_grams = f.total_grams[Picked]
```

## 3. Type mapping rules

| Your language | TypeScript | Why |
|---|---|---|
| `i8`–`i32`, `u8`–`u32`, `f64` | `number` | Fits exactly |
| `i64`, `u64` | `bigint` | `number` loses precision above 2^53 |
| `str` | `string` | |
| `distinct Sku` | Branded `string` | A plain string can't be passed as a `Sku` |
| `List[T]` | `readonly T[]` | Value semantics: mutating in TS can't affect the original |
| `Map[K, V]` | `ReadonlyMap<K, V>` | Same reason |
| `Option[T]` | `T \| null` | |
| `Result[T, E]` | `Result<T, E>` union | Errors stay explicit, no exceptions |
| Enum with fields | Discriminated union on `kind` | Exhaustive `switch` works in TS |
| Field with default | Optional in `…Input` type | Required in the output type |
| `Instant` | `Date` | |
| `uses clock` | Host-provided automatically | |
| `uses ffi`, `fs`, `process` | Compile error for browser targets | Capability not available there |

## 4. Generated `warehouse.d.ts`

```ts
// AUTO-GENERATED by eider. Do not edit.

export type Sku = string & { readonly __brand: "Sku" };
export function Sku(value: string): Sku;

export interface Money { readonly cents: bigint }

export interface Item {
  readonly sku: Sku;
  readonly name: string;
  readonly price: Money;
  readonly grams: number;
}

export interface Line { readonly sku: Sku; readonly qty: number }

export type BoxSize = "Small" | "Medium" | "Large";

export type OrderStatus =
  | { readonly kind: "Pending" }
  | { readonly kind: "Picking"; readonly picker: number; readonly started: Date }
  | { readonly kind: "Packed"; readonly box: BoxSize }
  | { readonly kind: "Shipped"; readonly tracking: string }
  | { readonly kind: "Cancelled"; readonly reason: string };

export interface Order {
  readonly id: bigint;
  readonly lines: readonly Line[];
  readonly status: OrderStatus;
  readonly priority: number;
}

// Defaults become optional on input
export interface OrderInput {
  id: bigint;
  lines: readonly Line[];
  status?: OrderStatus;
  priority?: number;
}
export function Order(input: OrderInput): Order;

export type Event =
  | { readonly kind: "StartPick"; readonly picker: number }
  | { readonly kind: "Pack" }
  | { readonly kind: "Ship"; readonly tracking: string }
  | { readonly kind: "Cancel"; readonly reason: string };

export type FulfillError =
  | { readonly kind: "UnknownSku"; readonly sku: Sku }
  | { readonly kind: "OutOfStock"; readonly sku: Sku; readonly wanted: number; readonly have: number }
  | { readonly kind: "TooHeavy"; readonly grams: number }
  | { readonly kind: "InvalidTransition"; readonly from: OrderStatus; readonly event: string };

export interface Inventory {
  readonly items: ReadonlyMap<Sku, Item>;
  readonly stock: ReadonlyMap<Sku, number>;
}

export type Result<T, E> =
  | { readonly ok: true; readonly value: T }
  | { readonly ok: false; readonly error: E };

export function advance(order: Order, event: Event, inv: Inventory): Result<Order, FulfillError>;
export function reserve_all(inv: Inventory, lines: readonly Line[]): Result<Inventory, FulfillError>;
export function choose_box(grams: number): Result<BoxSize, FulfillError>;

export function init(wasm?: URL | Response | BufferSource): Promise<void>;
```

Unit-only enums like `BoxSize` become string literal unions, which read more naturally in TS than `{ kind: "Small" }`.

## 5. Using it from TypeScript

```ts
import {
  init, Order, Sku, advance, reserve_all,
  type FulfillError, type Inventory,
} from "./warehouse.js";

await init();

const order = Order({
  id: 1001n,
  lines: [{ sku: Sku("A-100"), qty: 3 }],
});

function describe(e: FulfillError): string {
  switch (e.kind) {
    case "UnknownSku":        return `Unknown SKU ${e.sku}`;
    case "OutOfStock":        return `${e.sku}: wanted ${e.wanted}, have ${e.have}`;
    case "TooHeavy":          return `Too heavy: ${e.grams} g`;
    case "InvalidTransition": return `Can't ${e.event} from ${e.from.kind}`;
    // Missing a case is a TS compile error (with strict + noImplicitReturns)
  }
}

function fulfill(inv: Inventory) {
  const reserved = reserve_all(inv, order.lines);
  if (!reserved.ok) return console.error(describe(reserved.error));

  const picked = advance(order, { kind: "StartPick", picker: 7 }, reserved.value);
  if (!picked.ok) return console.error(describe(picked.error));

  console.log(picked.value.status.kind); // "Picking"
}

// Rejected by tsc: plain string is not a Sku
// reserve_all(inv, [{ sku: "A-100", qty: 1 }]);
```

## 6. Runtime validation at the boundary

TypeScript types are erased at runtime, so plain JS callers or `any` values could pass bad data. The generated `warehouse.js` validates every value entering WASM:

```ts
advance({ id: 5 } as any, { kind: "Pack" }, inv);
// TypeError: advance(order): order.id expected bigint, got number
//            order.lines is required
```

The checks are generated from the same type definitions, so they can never drift from the `.d.ts`.

## 7. Targets

| Target | Command | Use |
|---|---|---|
| Browser / any JS runtime | `--target wasm32 --emit ts` | Frontends, Deno, Bun, edge |
| Node native addon | `--target node-addon --emit ts` | Faster in Node, no WASM sandbox |
| Types only | `--emit ts-types` | API contracts with no shared logic |
